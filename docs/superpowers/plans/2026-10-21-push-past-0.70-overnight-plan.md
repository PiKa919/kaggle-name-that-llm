```python
# tests/test_length_gate.py
import numpy as np
import pytest
from src.config import VALID_18_MODELS, MODEL_TO_IDX
from src.eval_stats import length_stratified_balanced_accuracy
from src.hierarchical_classifier import HierarchicalAttributionHead, compute_typographic_priors

def test_length_stratified_metrics():
    y_true = np.array([0, 0, 1, 1, 2, 2])
    y_pred = np.array([0, 1, 1, 1, 2, 0])
    texts = [
        "word " * 45,   # bucket <60
        "word " * 50,   # bucket <60
        "word " * 75,   # bucket 60-90
        "word " * 80,   # bucket 60-90
        "word " * 150,  # bucket 120-180
        "word " * 160,  # bucket 120-180
    ]
    res = length_stratified_balanced_accuracy(y_true, y_pred, texts, bins=[0, 60, 90, 200])
    assert "0-60" in res
    assert "60-90" in res
    assert "90-200" in res

def test_fuse_with_length_gate():
    head = HierarchicalAttributionHead()
    N = 2
    logits_lab = np.zeros((N, 6))
    logits_siblings = {lab: np.zeros((N, count)) for lab, count in {"anthropic": 5, "google": 4, "openai": 5, "qwen": 1, "zai": 2, "moonshot": 1}.items()}
    texts = [
        "Short text with spaced em — dash.", # 6 words -> should get higher typo weight
        "This is a substantially longer text passage designed to test that standard length passages receive normal typographic weighting without distortion." # 20 words
    ]
    word_counts = np.array([6, 20])
    typo_priors = compute_typographic_priors(texts)
    P = head.fuse_with_length_gate(logits_lab, logits_siblings, typo_priors, word_counts, threshold_words=10)
    assert P.shape == (N, 18)
    assert np.allclose(P.sum(axis=1), 1.0, atol=1e-5)
```

```python
def test_pairwise_sibling_ensemble():
    N = 100
    D = 17
    # Synthetic quotient embeddings
    Z = np.random.randn(N, D)
    y = np.random.choice(len(VALID_18_MODELS), size=N)
    
    ensemble = PairwiseSiblingEnsemble(pairs=[("gemini-3.7-flash", "gemini-3.8-flash")])
    ensemble.fit(Z, y)
    corrections = ensemble.predict_log_odds_corrections(Z)
    assert ("gemini-3.7-flash", "gemini-3.8-flash") in corrections
    assert corrections[("gemini-3.7-flash", "gemini-3.8-flash")].shape == (N,)
```
