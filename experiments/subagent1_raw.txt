Key helper: robust normalization must handle NaN/inf.

```python
def _robust_zscore(raw, clip=8.0):
    X = np.asarray(raw, dtype=np.float64)
    if X.size == 0: return X.copy()
    med = np.median(X, axis=0)
    q25 = np.percentile(X, 25, axis=0)
    q75 = np.percentile(X, 75, axis=0)
    scale = (q75 - q25) / 1.349
    std = X.std(axis=0)
    scale = np.where(scale > 1e-12, scale, np.where(std > 1e-12, std, 1.0))
    Z = (X - med) / scale
    Z = np.nan_to_num(Z, nan=0.0, posinf=clip, neginf=-clip)
    return np.clip(Z, -clip, clip)
```

Quantile: average ranks per column scaled to [-1,1]; constant column → 0.

```python
def _quantile_scale(raw):
    X = np.asarray(raw, float)
    if X.size == 0: return X.copy()
    out = np.zeros_like(X)
    for j in range(X.shape[1]):
        col = X[:, j]
        order = np.argsort(col, kind="mergesort")
        ranks = np.empty(len(col), float)
        ranks[order] = np.arange(len(col), dtype=float)
        # average ties
        _, inv, counts = np.unique(col, return_inverse=True, return_counts=True)
        sums = np.zeros(len(counts)); np.add.at(sums, inv, ranks)
        ranks = (sums / counts)[inv]
        if len(col) > 1:
            out[:, j] = 2.0 * ranks / (len(col) - 1) - 1.0
    return out
```
